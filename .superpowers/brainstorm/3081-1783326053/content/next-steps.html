等拿到 spark-submit 后的完整流程

# Step 1: 打包 Python 环境（从你的 Docker 镜像）
docker run --rm -v $(pwd):/out your-notebook-image bash -c "
  python3 -m venv /tmp/pyspark_env && \
  source /tmp/pyspark_env/bin/activate && \
  pip install torch onnxruntime pandas numpy pyarrow && \
  tar -czf /out/pyspark_env.tar.gz -C /tmp pyspark_env"

# Step 2: 上传 HDFS
kinit -kt your.keytab your-user@REALM
hdfs dfs -mkdir -p /user/yourname/envs/
hdfs dfs -put pyspark_env.tar.gz /user/yourname/envs/

# Step 3: 提交 PySpark 任务
spark-submit \
  --master yarn --deploy-mode client \
  --queue your_queue \
  --num-executors 50 \
  --executor-cores 4 --executor-memory 16G \
  --archives hdfs:///user/yourname/envs/pyspark_env.tar.gz#venv \
  --conf spark.executorEnv.PYSPARK_PYTHON=./venv/bin/python \
  --conf spark.yarn.appMasterEnv.PYSPARK_PYTHON=./venv/bin/python \
  train.py
